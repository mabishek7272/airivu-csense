```typescript
import { EMBEDDING_DIM, bestMatch, normalise, type GalleryEntry } from '../src/faceMatch';

// A tiny deterministic PRNG (mulberry32) so "seed 1" always produces the same vector,
// the same role numpy's seeded default_rng plays in the Python test this mirrors
// (backend/tests/test_face_matching.py).
function mulberry32(seed: number) {
  let s = seed;
  return function () {
    s |= 0;
    s = (s + 0x6d2b79f5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function unit(seed: number): number[] {
  const rand = mulberry32(seed);
  const v: number[] = [];
  for (let i = 0; i < EMBEDDING_DIM; i++) {
    const u1 = Math.max(rand(), 1e-9);
    const u2 = rand();
    v.push(Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2));
  }
  return normalise(v);
}

function entry(name: string, seed: number): GalleryEntry {
  return { id: `id-${name}`, name, vector: unit(seed) };
}

describe('normalise', () => {
  it('rejects a zero vector', () => {
    expect(() => normalise(new Array(EMBEDDING_DIM).fill(0))).toThrow();
  });

  it('rejects the wrong dimension', () => {
    expect(() => normalise(new Array(10).fill(1))).toThrow();
  });

  it('produces a unit-length vector', () => {
    const v = normalise(unit(1).map((x) => x * 7.5));
    const length = Math.sqrt(v.reduce((sum, x) => sum + x * x, 0));

describe('bestMatch', () => {
```

```typescript
// mobile/apti/src/faceMatch.ts
/**
 * Cosine face matching, ported from `backend/shared/csense_shared/faces/matching.py`.
 * That module stores/decodes base64 templates for the real product's encrypted-at-rest
 * gallery; this one has no encoding step because embeddings here never leave app memory -
 * only `normalise` and the matching math are needed client-side.
 */

export const EMBEDDING_DIM = 512;
export const DEFAULT_THRESHOLD = 0.4;

export interface GalleryEntry {
  id: string;
  name: string;
  vector: number[]; // unit length, EMBEDDING_DIM long
}

export interface Match {
  id: string;
  name: string;
  similarity: number;
}

function dot(a: number[], b: number[]): number {
  let sum = 0;
  for (let i = 0; i < a.length; i++) sum += a[i] * b[i];
  return sum;
}

/** ai-runtime's `/internal/v1/infer-face` already L2-normalises its embeddings; this
 *  re-normalises defensively rather than trusting that on every caller. */
export function normalise(vector: number[]): number[] {
  if (vector.length !== EMBEDDING_DIM) {
    throw new Error(`expected a ${EMBEDDING_DIM}-d embedding, got ${vector.length}`);
  }
  const length = Math.sqrt(dot(vector, vector));
  if (!Number.isFinite(length) || length === 0) {
    throw new Error('embedding has zero length or is not finite');
  }
  return vector.map((v) => v / length);
}

/** The single most similar gallery entry at or above `threshold`, or null. An identity
 *  may appear several times (one row per registered photo); the entry with the highest
 *  score wins, exactly like the Python `best_match` this mirrors. */
export function bestMatch(
  embedding: number[],
  gallery: GalleryEntry[],
  threshold: number = DEFAULT_THRESHOLD,
): Match | null {
  if (gallery.length === 0) return null;
  const query = normalise(embedding);
  let best: Match | null = null;
  for (const candidate of gallery) {
    const similarity = dot(query, candidate.vector);
    if (best === null || similarity > best.similarity) {
      best = { id: candidate.id, name: candidate.name, similarity };
    }
  }
  if (best === null || best.similarity < threshold) return null;
  return { ...best, similarity: Math.round(best.similarity * 10000) / 10000 };
}
```

```typescript
// mobile/apti/src/tamperHeuristic.ts
/**
 * Flags a likely covered/blocked camera by watching the encoded JPEG's own byte size,
 * not raw pixels - no RN API in this app's dependency set exposes raw pixel stats without
 * a much heavier native module. A blocked lens (a finger, a black covering) is nearly
 * uniform colour, which JPEG compresses far smaller than a normal scene. A rolling
 * average avoids a false positive on one naturally simple frame (e.g. a blank wall) and
 * only fires once enough history exists to trust the average.
 */

const WINDOW = 8;
const MIN_SAMPLES = 4;
const DROP_RATIO = 0.35;

export class TamperDetector {
  private samples: number[] = [];

  /** Call once per captured frame with its encoded JPEG size in bytes. Returns true if
   *  this frame looks like a covered/blocked lens relative to recent frames. */
  check(byteSize: number): boolean {
    const tampered = this.samples.length >= MIN_SAMPLES && byteSize < this.average() * DROP_RATIO;
    this.samples.push(byteSize);
    if (this.samples.length > WINDOW) this.samples.shift();
    return tampered;
  }

  reset(): void {
    this.samples = [];
  }

  private average(): number {
    return this.samples.reduce((a, b) => a + b, 0) / this.samples.length;
  }
}
```

```typescript
// mobile/apti/src/history.ts
import AsyncStorage from '@react-native-async-storage/async-storage';
import { Directory, File, Paths } from 'expo-file-system';

export type HistorySource = 'detect' | 'monitor';
export type Severity = 'low' | 'medium' | 'high';

export interface HistoryEntry {
  id: string;
  source: HistorySource;
  label: string;
  confidence: number | null;
  severity: Severity;
  timestamp: number;
  /** A local file:// path only - never uploaded, never synced. */
  thumbnailUri: string | null;
}

export const MAX_HISTORY_ENTRIES = 200;
const STORAGE_KEY = 'apti.history.v1';
const historyDir = new Directory(Paths.document, 'apti-history');

/** Pure: applies the cap, returning what to keep and what was dropped (so the caller can
 *  clean up dropped thumbnails). Newest-first ordering by `timestamp`. */
export function applyCap(
  entries: HistoryEntry[],
  max: number = MAX_HISTORY_ENTRIES,
): { kept: HistoryEntry[]; dropped: HistoryEntry[] } {
  if (entries.length <= max) return { kept: entries, dropped: [] };
  const sorted = [...entries].sort((a, b) => b.timestamp - a.timestamp);
  return { kept: sorted.slice(0, max), dropped: sorted.slice(max) };
}

export async function loadHistory(): Promise<HistoryEntry[]> {
  const raw = await AsyncStorage.getItem(STORAGE_KEY);
  if (!raw) return [];
  try {
    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed : [];
  } catch {
    return [];
  }
}

async function saveHistory(entries: HistoryEntry[]): Promise<void> {
  await AsyncStorage.setItem(STORAGE_KEY, JSON.stringify(entries));
}

/** Copies a temporary camera/picker file into this app's own document directory under a
 *  name tied to this entry, so it survives after the OS cleans up the source's temp file -
 *  still entirely local to the device. A failure here is not fatal; the entry still saves
 *  without a thumbnail. */
async function saveThumbnail(id: string, sourceUri: string): Promise<string | null> {
  try {
    if (!historyDir.exists) historyDir.create({ intermediates: true });
    const source = new File(sourceUri);
    const dest = new File(historyDir, `${id}.jpg`);
    await source.copy(dest);
    return dest.uri;
  } catch {
    return null;
  }
}

function deleteThumbnail(uri: string | null): void {
  if (!uri) return;
  try {
    const file = new File(uri);
    if (file.exists) file.delete();
  } catch {
    // best-effort cleanup
  }
}

export async function addHistoryEntry(input: {
  source: HistorySource;
  label: string;
  confidence: number | null;
  severity: Severity;
  /** A local frame/photo uri to copy in as this entry's thumbnail, if any. */
  thumbnailSourceUri?: string;
}): Promise<HistoryEntry> {
  const id = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
  const thumbnailUri = input.thumbnailSourceUri ? await saveThumbnail(id, input.thumbnailSourceUri) : null;
  const entry: HistoryEntry = {
    id,
    source: input.source,
    label: input.label,
    confidence: input.confidence,
    severity: input.severity,
    timestamp: Date.now(),
    thumbnailUri,
  };
  const current = await loadHistory();
  const { kept, dropped } = applyCap([entry, ...current]);
  dropped.forEach((d) => deleteThumbnail(d.thumbnailUri));
  await saveHistory(kept);
  return entry;
}

export async function clearHistory(): Promise<void> {
  const current = await loadHistory();
  current.forEach((e) => deleteThumbnail(e.thumbnailUri));
  await AsyncStorage.removeItem(STORAGE_KEY);
}
```

```typescript
// mobile/apti/src/FaceGalleryContext.tsx
import React, { createContext, useCallback, useContext, useMemo, useState } from 'react';
import { bestMatch, type GalleryEntry, type Match } from './faceMatch';

interface FaceGalleryValue {
  entries: GalleryEntry[];
  /** Distinct registered people (an identity may have several template rows). */
  personCount: number;
  /** Registers one captured photo's embedding under `name`, sharing `id` across a
   *  person's several photos - mirrors the real product's "several templates, one
   *  identity" gallery shape, so matching picks whichever template scores best. */
  register: (id: string, name: string, vector: number[]) => void;
  /** A stricter check than `match` - used by the registration wizard to warn "this looks
   *  like someone already registered" before saving a duplicate. */
  findDuplicate: (vector: number[]) => Match | null;
  match: (vector: number[]) => Match | null;
  clear: () => void;
}

const FaceGalleryContext = createContext<FaceGalleryValue | null>(null);

// Stricter than the 0.4 matching threshold - a warning, not the pass/fail bar for "is
// this the same person" during live detection.
const DUPLICATE_THRESHOLD = 0.6;

export function FaceGalleryProvider({ children }: { children: React.ReactNode }) {
  const [entries, setEntries] = useState<GalleryEntry[]>([]);

  const register = useCallback((id: string, name: string, vector: number[]) => {
    setEntries((prev) => [...prev, { id, name, vector }]);
  }, []);

  const findDuplicate = useCallback((vector: number[]) => bestMatch(vector, entries, DUPLICATE_THRESHOLD), [entries]);
  const match = useCallback((vector: number[]) => bestMatch(vector, entries), [entries]);
  const clear = useCallback(() => setEntries([]), []);

  const value = useMemo<FaceGalleryValue>(
    () => ({
      entries,
      personCount: new Set(entries.map((e) => e.id)).size,
      register,
      findDuplicate,
      match,
      clear,
    }),
    [entries, register, findDuplicate, match, clear],
  );

  return <FaceGalleryContext.Provider value={value}>{children}</FaceGalleryContext.Provider>;
}

export function useFaceGallery(): FaceGalleryValue {
  const ctx = useContext(FaceGalleryContext);
  if (!ctx) throw new Error('useFaceGallery must be used within a FaceGalleryProvider');
  return ctx;
}
```

```python
async def detect_face(request: Request, image: UploadFile = File(...)):
    if image.content_type not in ALLOWED_TYPES:
        return _error(415, "unsupported_image", "Send a JPEG, PNG or WebP image.")

    try:
        await check_and_increment(
            request.app.state.redis, settings, client_id=_client_key(request),
            limit_per_minute=PER_IP_LIMIT_PER_MINUTE,
        )
    except RateLimitExceededError as exc:
        return _error(429, "rate_limited", "Too many requests. Slow down a little.",
                      {"Retry-After": str(exc.retry_after_seconds)})

    data = await image.read(MAX_IMAGE_BYTES + 1)
    if not data:
        return _error(400, "empty_image", "No image received.")
    if len(data) > MAX_IMAGE_BYTES:
        return _error(413, "image_too_large", "Image is larger than 1.5 MB. Send a smaller photo.")

    slots: asyncio.Semaphore = request.app.state.slots
    if slots.locked():
        return _error(503, "busy", "The demo is busy. Try again in a moment.", {"Retry-After": "1"})

    async with slots:
        started = time.monotonic()
        try:
            response = await request.app.state.http.post(
                "/internal/v1/infer-face",
                data={"confidence": "0.5"},
                files={"frame": ("frame.jpg", data, image.content_type)},
            )
        except httpx.HTTPError:
            logger.warning("demo_infer_face_unreachable")
            return _error(502, "inference_unavailable", "Detection is unavailable right now.")

    if response.status_code == 400:
        return _error(400, "bad_image", "That image could not be read.")
    if response.status_code != 200:
        logger.warning("demo_infer_face_failed", extra={"status": response.status_code})
        return _error(502, "inference_failed", "Detection failed. Try again.")

    payload = response.json()
    width, height = payload.get("frame_size", [0, 0])
    faces = [
        FaceOut(
            box=BoxOut(x1=f["bbox"][0], y1=f["bbox"][1], x2=f["bbox"][2], y2=f["bbox"][3]),
            detector_confidence=float(f["detector_confidence"]),
            embedding=f["embedding"],
        )
        for f in payload.get("faces", [])
    ]
    logger.info("demo_detect_face", extra={
        "faces": len(faces), "ms": round((time.monotonic() - started) * 1000)})
    return DetectFaceOut(width=width, height=height,
                          inference_ms=float(payload.get("inference_ms", 0)), faces=faces)
```

```typescript
export interface FaceDetection {
  box: NormBox;
  detectorConfidence: number;
  embedding: number[];
}

export interface DetectFaceResult {
  width: number;
  height: number;
  inference_ms: number;
  faces: FaceDetection[];
}

export async function detectFaces(imageUri: string, signal?: AbortSignal): Promise<DetectFaceResult> {
  const form = new FormData();
  form.append('image', { uri: imageUri, name: 'frame.jpg', type: 'image/jpeg' } as unknown as Blob);

  let res: Response;
  try {
    res = await fetch(`${API_BASE}/api/v1/demo/detect-face`, { method: 'POST', body: form, signal });
  } catch (error) {
    if ((error as Error).name === 'AbortError') throw error;
    throw new DemoError('No connection to the demo service.');
  }

  if (res.status === 429 || res.status === 503) {
    const seconds = Number(res.headers.get('retry-after') ?? '1');
    throw new Throttled(Math.max(seconds, 1) * 1000);
  }
  if (!res.ok) {
    const body = await res.json().catch(() => null);
    throw new DemoError(body?.message ?? 'Face detection failed. Try again.');
  }
  const payload = await res.json();
  return {
    width: payload.width,
    height: payload.height,
    inference_ms: payload.inference_ms,
    faces: (payload.faces ?? []).map((f: { box: NormBox; detector_confidence: number; embedding: number[] }) => ({
      box: f.box,
      detectorConfidence: f.detector_confidence,
      embedding: f.embedding,
    })),
  };
}
```

```typescript
          if (people > 1) pushEvent('multiple_people', `${people} people in frame`, 'medium');

          const phone = objects.detections.find((d) => d.label === 'Cell phone');
          if (phone) pushEvent('phone', 'Mobile phone detected', 'high');

          for (const face of faces.faces) {
            const match = gallery.match(face.embedding);
            if (!match) pushEvent('unknown_face', 'Unknown face detected', 'medium');
          }
        } catch {
          // A single failed cycle is not fatal - the next one just tries again.
        }
        const wait = CYCLE_MS - (Date.now() - started);
        await new Promise((resolve) => setTimeout(resolve, Math.max(wait, 200)));
      }
    })();

    return () => {
      cancelled = true;
    };
  }, [running, permission?.granted]);

  function start() {
    tamper.reset();
    setEvents([]);
    setElapsedSec(0);
    setRunning(true);
  }

  function stop() {
    setRunning(false);
  }

  const pad = (n: number) => String(n).padStart(2, '0');
  const mm = pad(Math.floor(elapsedSec / 60));
  const ss = pad(elapsedSec % 60);

  return (
    <View style={styles.root}>
      <View style={styles.header}>
        <Pressable onPress={onBack} accessibilityRole="button" accessibilityLabel="Back">
          <Icon name="arrow-left" />
        </Pressable>
        <Kicker>Live Monitoring</Kicker>
      </View>

      {!permission?.granted ? (
        <View style={styles.centered}>
          <Text style={{ color: color.text, fontSize: 14, textAlign: 'center', marginBottom: 12 }}>
            Camera access is needed to monitor.
          </Text>
          <Btn variant="primary" onPress={requestPermission}>
            Allow camera
          </Btn>
        </View>
      ) : (
        <>
          <View style={styles.stage}>
            <CameraView ref={cameraRef} style={StyleSheet.absoluteFill} facing="back" />
            {running && (
              <View style={styles.statusBar}>
                <Text style={styles.statusText}>{mm}:{ss}</Text>
                <Text style={styles.statusText}>{events.length} alert{events.length === 1 ? '' : 's'}</Text>
              </View>
            )}
          </View>

          <Btn variant="primary" block onPress={running ? stop : start}>
            {running ? 'Stop' : 'Run'}
          </Btn>

          {!running && events.length > 0 && (
            <View style={styles.summary}>
              <Text style={{ color: color.text, fontFamily: fontHeading, fontSize: 15 }}>Session summary</Text>
              <Text style={{ color: textMuted, fontSize: 13 }}>
                {mm}:{ss} · {events.length} alert{events.length === 1 ? '' : 's'}
              </Text>
            </View>
          )}

          <FlatList

            keyExtractor={(e) => e.id}

            )}
          />
        </>
      )}
    </View>
  );
}
```
